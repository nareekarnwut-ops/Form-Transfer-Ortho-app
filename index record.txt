<script>
  // โหลดจำนวนครั้งการลงทะเบียนเมื่อเปิดหน้าเว็บ
  window.addEventListener('DOMContentLoaded', () => {
    // กำหนดวันที่ปัจจุบันเป็นค่าเริ่มต้นในช่อง Date
    const today = new Date().toISOString().split('T')[0];
    document.getElementById('dateRequest').value = today;
    document.getElementById('dateReceive').value = today;

    // ดึงจำนวนรายการทั้งหมด
    google.script.run
      .withSuccessHandler(updateCounter)
      .getTotalRecordsCount();
  });

  function updateCounter(count) {
    document.getElementById('totalRecords').innerText = count;
  }

  // ส่งข้อมูลฟอร์ม
  document.getElementById('transferForm').addEventListener('submit', function(e) {
    e.preventDefault();

    const submitBtn = document.getElementById('submitBtn');
    const alertBox = document.getElementById('alertBox');
    
    submitBtn.disabled = true;
    submitBtn.innerText = 'กำลังบันทึกข้อมูล...';
    alertBox.style.display = 'none';

    const formData = {
      dateRequest: document.getElementById('dateRequest').value,
      dateReceive: document.getElementById('dateReceive').value,
      hn: document.getElementById('hn').value.trim(),
      name: document.getElementById('name').value.trim(),
      diagnosis: document.getElementById('diagnosis').value.trim(),
      doctor: document.getElementById('doctor').value.trim(),
      ward: document.getElementById('ward').value.trim()
    };

    google.script.run
      .withSuccessHandler(function(response) {
        submitBtn.disabled = false;
        submitBtn.innerText = 'ลงทะเบียนรับย้าย';

        if (response.status === 'success') {
          showAlert(response.message, 'success');
          // อัปเดตตัวเลขจำนวนการลงทะเบียน
          updateCounter(response.totalRecords);
          
          // ล้างข้อมูลในฟอร์ม (คงวันที่ไว้)
          document.getElementById('hn').value = '';
          document.getElementById('name').value = '';
          document.getElementById('diagnosis').value = '';
          document.getElementById('doctor').value = '';
          document.getElementById('ward').value = '';
        } else {
          showAlert('เกิดข้อผิดพลาด: ' + response.message, 'error');
        }
      })
      .withFailureHandler(function(err) {
        submitBtn.disabled = false;
        submitBtn.innerText = 'ลงทะเบียนรับย้าย';
        showAlert('เกิดข้อผิดพลาดในการเชื่อมต่อ: ' + err, 'error');
      })
      .savePatientData(formData);
  });

  function showAlert(message, type) {
    const alertBox = document.getElementById('alertBox');
    alertBox.innerText = message;
    alertBox.className = 'alert alert-' + type;
    alertBox.style.display = 'block';
  }
</script>
